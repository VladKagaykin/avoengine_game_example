fn floor_i(x) {
    let xi = x.to_int();
    if x < 0.0 {
        if xi.to_float() != x {
            xi = xi - 1;
        }
    }
    return xi;
}

fn hash2d(x, z, s) {
    let xi = x & 0xfffff;
    let zi = z & 0xfffff;
    let h = s;
    h = h ^ (xi * 374761393 + zi * 668265263);
    h = h & 0x7fffffff;
    h = (h ^ (h >> 13)) * 1274126177;
    h = h & 0x7fffffff;
    h = h ^ (h >> 16);
    return h & 0x7fffffff;
}

fn make_floor_vertices(x, z, size) {
    let x1 = x + size;
    let z1 = z + size;
    return [
        x, 0.0, z,    x1, 0.0, z,    x1, 0.0, z1,
        x, 0.0, z,    x1, 0.0, z1,   x, 0.0, z1
    ];
}

fn make_ceiling_vertices(x, z, size, y) {
    let x1 = x + size;
    let z1 = z + size;
    return [
        x, y, z,    x1, y, z,    x1, y, z1,
        x, y, z,    x1, y, z1,   x, y, z1
    ];
}

fn push_floor(x, z, size, color, chunk_name) {
    let empty_tex = \"\";
    let verts = make_floor_vertices(x, z, size);
    let comp = Draw_components(\"3d_object\", 0.0, 0.0, 0.0, ' ', verts, color, empty_tex, chunk_name);
    push_static_draw(comp);
}

fn push_ceiling(x, z, size, height, color, chunk_name) {
    let empty_tex = \"\";
    let verts = make_ceiling_vertices(x, z, size, height);
    let comp = Draw_components(\"3d_object\", 0.0, 0.0, 0.0, ' ', verts, color, empty_tex, chunk_name);
    push_static_draw(comp);
}

fn push_wall_x(x1, x2, z0, y_top, thickness, color, chunk_name) {
    let empty_tex = \"\";
    let verts = make_wall_box_x(x1, x2, z0, y_top, thickness);
    let comp = Draw_components(\"3d_object\", 0.0, 0.0, 0.0, '#', verts, color, empty_tex, chunk_name);
    push_static_draw(comp);

fn push_wall_z(x0, z1, z2, y_top, thickness, color, chunk_name) {

fn generate_chunk(cx, cz, seed, chunk_size, cell_size, wall_height, wall_threshold, wall_thickness, wall_color, floor_color, ceiling_color, cam_x, cam_z) {
    let base_x = cx * chunk_size * cell_size;
    let base_z = cz * chunk_size * cell_size;

    let chunk_name = \"chunk_\" + cx.to_string() + \"_\" + cz.to_string();

    let lx = 0;
    while lx < chunk_size {
        let lz = 0;
        while lz < chunk_size {
            let wx = base_x + lx.to_float() * cell_size;
            let wz = base_z + lz.to_float() * cell_size;

            let ax = cx * chunk_size + lx;
            let az = cz * chunk_size + lz;

            push_floor(wx, wz, cell_size, floor_color, chunk_name);
            push_ceiling(wx, wz, cell_size, wall_height, ceiling_color, chunk_name);

            let noise_upper = hash2d(ax, az - 1, seed).to_float() / 2147483647.0;
            if noise_upper > wall_threshold {
                let x1 = wx;
                let x2 = wx + cell_size;
                push_wall_x(x1, x2, wz, wall_height, wall_thickness, wall_color, chunk_name);
            }

            let noise_left = hash2d(ax - 1, az, seed).to_float() / 2147483647.0;
            if noise_left > wall_threshold {
                let z1 = wz;
                let z2 = wz + cell_size;
                push_wall_z(wx, z1, z2, wall_height, wall_thickness, wall_color, chunk_name);
            }

            lz = lz + 1;
        }
        lx = lx + 1;
    }
}

fn join_array(arr, sep) {
    let result = \"\";
    let len = arr.len();
    if len == 0 { return result; }
    result = arr[0].to_string();
    let i = 1;
    while i < len {
        result += sep + arr[i].to_string();
        i = i + 1;
    }
    return result;
}

let chunk_temp = chunk_size * cell_size;
let chunk_x = floor_i(cam_x / chunk_temp);
let chunk_z = floor_i(cam_z / chunk_temp);

let target = [];
let dx = -1;
while dx <= 1 {
    let dz = -1;
    while dz <= 1 {
        let name = (chunk_x + dx).to_string() + \"_\" + (chunk_z + dz).to_string();
        target.push(name);
        dz = dz + 1;
    }
    dx = dx + 1;
}

let loaded_str = get_map_object(\"loaded_chunks\");
let loaded = [];
if loaded_str != \"\" && loaded_str != \"none\" {
    let parts = loaded_str.split(\",\");
    let i = 0;
    while i < parts.len() {
        loaded.push(parts[i]);
        i = i + 1;
    }
}

let changed = false;

let i = 0;
while i < loaded.len() {
    let name = loaded[i];
    if !target.contains(name) {
        let full_name = \"chunk_\" + name;
        let removed = remove_static_draw_by_name(full_name);
        if removed > 0 {
            changed = true;
        }
    }
    i = i + 1;
}

let i = 0;
while i < target.len() {
    let name = target[i];
    if !loaded.contains(name) {
        let parts = name.split(\"_\");
        let gx = parts[0].to_int();
        let gz = parts[1].to_int();
        generate_chunk(gx, gz, seed, chunk_size, cell_size, wall_height, wall_threshold, wall_thickness, wall_color, floor_color, ceiling_color, cam_x, cam_z);
        changed = true;
    }
    i = i + 1;
}

    let new_loaded = join_array(target, \",\");
    set_map_object(\"loaded_chunks\", new_loaded);
    set_scene_changed(true);
}
